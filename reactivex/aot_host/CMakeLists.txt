# Track transitive local imports, including macros. An adapter edit must not
# regenerate/recompile every unrelated core Rx test.
function(local_dependencies script output)
  set(pending "${script}")
  set(seen)
  while(pending)
    list(POP_FRONT pending current)
    get_filename_component(current "${current}" ABSOLUTE)
    if(current IN_LIST seen)
      continue()
    endif()
    list(APPEND seen "${current}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${current}")
    get_filename_component(parent "${current}" DIRECTORY)
    file(STRINGS "${current}" imports REGEX "^require +[.]")
    foreach(import IN LISTS imports)
      string(REGEX REPLACE "^require +([^ ]+).*" "\\1" name "${import}")
      get_filename_component(dependency "${parent}/${name}" ABSOLUTE)
      if(EXISTS "${dependency}")
        list(APPEND pending "${dependency}")
      else()
        message(FATAL_ERROR "Missing local import ${dependency}")
      endif()
    endforeach()
  endwhile()
  set(${output} ${seen} PARENT_SCOPE)
endfunction()
set(generated)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
foreach(script IN LISTS scripts)
  local_dependencies("${script}" dependencies)
  get_filename_component(stem "${script}" NAME_WE)
  set(cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/${stem}.cpp")
  add_custom_command(OUTPUT "${cpp}"
    COMMAND $<TARGET_FILE:DAS::daslang> -aot "${script}" "${cpp}"
    DEPENDS "${script}" ${dependencies} VERBATIM)
  list(APPEND generated "${cpp}")
endforeach()
